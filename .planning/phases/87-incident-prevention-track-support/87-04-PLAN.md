     * Idempotent seeder for the Incident Prevention track.
     * Used by:
     *   - POST /api/projects (new-project flow when active_tracks.incident_prevention === true)
     *   - PATCH /api/projects/[projectId]/settings (retroactive seeding when track flips false→true)
     *
     * Safe to call multiple times for the same project — every insert uses
     * onConflictDoNothing() or a WHERE NOT EXISTS guard.
     *
     * @param tx Drizzle transaction handle (caller must wrap in db.transaction(...))
     * @param projectId Target project ID
     */
    export async function seedIncidentPreventionForProject(
      tx: PgTransaction<any, any, any>,
      projectId: number,
    ): Promise<void> {
      // 1. arch_track — guard with select-then-insert
      const existing = await tx.select({ id: archTracks.id })
        .from(archTracks)
        .where(and(eq(archTracks.project_id, projectId), eq(archTracks.name, 'Incident Prevention Track')))
        .limit(1)

      let ipTrackId: number
      if (existing.length === 0) {
        const [inserted] = await tx.insert(archTracks)
          .values({ project_id: projectId, name: 'Incident Prevention Track', display_order: 30 })
          .returning({ id: archTracks.id })
        ipTrackId = inserted.id
      } else {
        ipTrackId = existing[0].id
      }

      // 2. Section arch_nodes — use onConflictDoNothing if unique index on (project_id, track_id, name) exists,
      //    otherwise select-then-insert pattern. Verify by reading db/schema.ts for archNodes uniqueIndex.
      const seedSection = async (name: string, display_order: number) => {
        const existing = await tx.select({ id: archNodes.id })
          .from(archNodes)
          .where(and(
            eq(archNodes.project_id, projectId),
            eq(archNodes.track_id, ipTrackId),
            eq(archNodes.name, name),
          ))
          .limit(1)
        if (existing.length > 0) return existing[0].id
        const [inserted] = await tx.insert(archNodes)
          .values({
            project_id: projectId, track_id: ipTrackId, name,
            display_order, status: 'planned', node_type: 'section', source_trace: 'retroactive-seed',
          })
          .returning({ id: archNodes.id })
        return inserted.id
      }

      const sectionDiId = await seedSection('Data Ingestion', 10)
      const sectionReId = await seedSection('Risk Engine', 20)
      const sectionDwId = await seedSection('Decision & Write-Back', 30)

      // 3. Change Risk Console (parent_id=null, node_type='console')
      const consoleExisting = await tx.select({ id: archNodes.id })
        .from(archNodes)
        .where(and(
          eq(archNodes.project_id, projectId),
          eq(archNodes.track_id, ipTrackId),
          eq(archNodes.name, 'Change Risk Console'),
        ))
        .limit(1)
      if (consoleExisting.length === 0) {
        await tx.insert(archNodes).values({
          project_id: projectId, track_id: ipTrackId, name: 'Change Risk Console',
          display_order: 15, status: 'planned', node_type: 'console', source_trace: 'retroactive-seed',
        })
      }

      // 4. Sub-capability arch_nodes (4 + 5 + 4 = 13)
      const subCaps: { parent: number; name: string; order: number }[] = [
        // Data Ingestion (4)

        { parent: sectionDiId, name: 'CMDB Connectors',                  order: 2 },

        { parent: sectionDwId, name: 'ITSM Write-Back (ServiceNow / JSM)', order: 2 },
        { parent: sectionDwId, name: 'CAB Notifications',                order: 3 },
        { parent: sectionDwId, name: 'Reporting & Dashboards',           order: 4 },
      ]
      for (const sc of subCaps) {
        const existing = await tx.select({ id: archNodes.id })
          .from(archNodes)
          .where(and(
            eq(archNodes.project_id, projectId),
            eq(archNodes.track_id, ipTrackId),
            eq(archNodes.parent_id, sc.parent),
            eq(archNodes.name, sc.name),
          ))
          .limit(1)
        if (existing.length === 0) {
          await tx.insert(archNodes).values({
            project_id: projectId, track_id: ipTrackId, parent_id: sc.parent, name: sc.name,
            display_order: sc.order, status: 'planned', node_type: 'sub-capability', source_trace: 'retroactive-seed',
          })
        }
      }

      // 5. WBS L1 items (10) — use ON CONFLICT DO NOTHING if wbs_items has a uniq index on (project_id, track, name, parent_id),
      //    otherwise select-then-insert. Read db/schema.ts wbsItems to verify before choosing pattern.
      const wbsL1: { name: string; display_order: number }[] = [
        { name: 'Kickoff & Scope',                       display_order: 1 },
        { name: 'Change Process Discovery',              display_order: 2 },
        { name: 'Single Sign-On',                        display_order: 3 },
        { name: 'ITSM Integration (ServiceNow / JSM)',   display_order: 4 },
        { name: 'Data Source Connectors',                display_order: 5 },
        { name: 'Risk Categories & Weights',             display_order: 6 },

            eq(wbsItems.project_id, projectId),
            eq(wbsItems.track, 'Incident Prevention'),
            sql`${wbsItems.parent_id} IS NULL`,
            eq(wbsItems.name, l1.name),
          ))
          .limit(1)
        if (existing.length === 0) {
          const [inserted] = await tx.insert(wbsItems).values({
            project_id: projectId, track: 'Incident Prevention', name: l1.name,
            display_order: l1.display_order, level: 1, parent_id: null, status: 'not_started',
          }).returning({ id: wbsItems.id })
          l1IdMap.set(l1.name, inserted.id)
        } else {
          l1IdMap.set(l1.name, existing[0].id)
        }
      }

      // 6. WBS L2 sub-tasks (~33) — see <interfaces> RECOMMENDED set
      const wbsL2: { parentName: string; subs: string[] }[] = [
        { parentName: 'Kickoff & Scope',                     subs: ['Stakeholder Identification', 'Success Criteria Definition', 'Change Volume Baseline'] },

        { parentName: 'Go-Live & CAB Enablement',            subs: ['CAB Training Sessions', 'Score Interpretation Guide', 'Production Cutover', 'Post-Launch Review'] },
      ]
      for (const group of wbsL2) {
        const parentId = l1IdMap.get(group.parentName)
        if (!parentId) continue
        for (let i = 0; i < group.subs.length; i++) {
          const subName = group.subs[i]
          const existing = await tx.select({ id: wbsItems.id })
            .from(wbsItems)
            .where(and(
              eq(wbsItems.project_id, projectId),
              eq(wbsItems.track, 'Incident Prevention'),
              eq(wbsItems.parent_id, parentId),
              eq(wbsItems.name, subName),
            ))
            .limit(1)
          if (existing.length === 0) {
            await tx.insert(wbsItems).values({
              project_id: projectId, track: 'Incident Prevention', name: subName,
              display_order: i + 1, level: 2, parent_id: parentId, status: 'not_started',
            })
          }
        }
      }

      // 7. Onboarding phases + steps from INCIDENT_PREVENTION_ONBOARDING_CONFIG
      for (const phase of INCIDENT_PREVENTION_ONBOARDING_CONFIG) {
        // Phase: assume onboarding_phases has uniq index on (project_id, track, name) — use onConflictDoNothing
        const [phaseRow] = await tx.insert(onboardingPhases)
          .values({ project_id: projectId, track: 'Incident Prevention', name: phase.name, display_order: phase.display_order })
          .onConflictDoNothing()
          .returning({ id: onboardingPhases.id })
        let phaseId: number
        if (phaseRow) {
          phaseId = phaseRow.id
        } else {
          const [existingPhase] = await tx.select({ id: onboardingPhases.id })
            .from(onboardingPhases)
            .where(and(
              eq(onboardingPhases.project_id, projectId),
              eq(onboardingPhases.track, 'Incident Prevention'),
              eq(onboardingPhases.name, phase.name),
            ))
            .limit(1)
          phaseId = existingPhase.id
        }

        for (let i = 0; i < phase.steps.length; i++) {
          await tx.insert(onboardingSteps).values({
            phase_id: phaseId, name: phase.steps[i], display_order: i + 1, status: 'not_started',

        }

      await tx.insert(teamOnboardingStatus).values({
        project_id: projectId, team_name: 'Team Gamma', track: 'Incident Prevention', source: 'template',
