<!-- Slack callback token exchange (source: RESEARCH.md) -->
```typescript
// POST https://slack.com/api/oauth.v2.access
// Response: { authed_user: { access_token: 'xoxp-...', scope: 'search:read' } }
// NO refresh_token from Slack — store access_token in BOTH access_token AND refresh_token columns
// (refresh_token is NOT NULL in schema — use access_token as placeholder)
const form = new URLSearchParams({ client_id, client_secret, code, redirect_uri });
const resp = await fetch('https://slack.com/api/oauth.v2.access', {
  method: 'POST',
  headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
  body: form.toString(),
});
const data = await resp.json();
const token = data.authed_user?.access_token;
```
