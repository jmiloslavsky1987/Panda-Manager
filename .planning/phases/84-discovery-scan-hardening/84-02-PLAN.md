type SlackAdapterInput = UserSourceToken | { token: string; channels: string[] };

// Distinguish in constructor: UserSourceToken has 'id' field; legacy has 'channels' field
constructor(input: SlackAdapterInput) {
  if ('channels' in input) {
    // Legacy bot token path
    this.token = input.token;
    this.mode = 'legacy';
    this.channels = input.channels;
  } else {
    // User OAuth token path
    this.token = input.access_token ?? '';
    this.mode = 'oauth';
    this.channels = [];
  }
}
