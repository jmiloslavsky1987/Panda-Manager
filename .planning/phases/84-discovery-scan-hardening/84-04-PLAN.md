<!-- arch_node approve case (from RESEARCH.md code examples) -->
```typescript
case 'arch_node': {
  const { name, track_name } = JSON.parse(item.content) as { name: string; track_name: string };
  await db.transaction(async (tx) => {
    let [track] = await tx.select({ id: archTracks.id })
      .from(archTracks)
      .where(and(eq(archTracks.project_id, projectId), eq(archTracks.name, track_name)));
    if (!track) {
      [track] = await tx.insert(archTracks)
        .values({ project_id: projectId, name: track_name, display_order: 0 })
        .returning();
    }
    await tx.insert(archNodes).values({
      project_id: projectId,
      track_id: track.id,
      name,
      display_order: 999,
      node_type: 'sub-capability',
    }).onConflictDoNothing(); // unique index: (project_id, track_id, name)
  });
  break;
}
```
